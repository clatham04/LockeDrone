# signal_controller_PC — Python dependencies
# Install in your virtual environment:  pip install -r requirements.txt
#
# This pulls the CUDA (GPU) build of PyTorch so the pose model runs fast on an NVIDIA card
# (e.g. RTX 3080). The startup line of run.py / detect_test.py will say "on GPU (CUDA)".
#
# >> NO NVIDIA GPU? Delete the next 3 lines (the --extra-index-url and the two +cu124 lines).
#    ultralytics will then install the regular CPU build of torch automatically.

--extra-index-url https://download.pytorch.org/whl/cu124
torch==2.5.1+cu124          # GPU PyTorch, CUDA 12.4
torchvision==0.20.1+cu124   # must match the torch version

# --- Camera (drone_camera.py / detect_test.py) ---
opencv-python           # cv2 — pull/decode the RTSP stream
numpy                   # frame buffer handling

# --- Human detection (detect_test.py / follow_human.py) ---
ultralytics             # YOLO person + pose detection
pyyaml                  # read model metadata.yaml

# --- Flight controls (signals_control.py) ---
keyboard                # Q = land, SPACE = emergency stop hotkeys (run terminal as Administrator)
